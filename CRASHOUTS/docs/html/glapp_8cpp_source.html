59bool GLApp::init(GLint w, GLint h, std::string t) {
60 GLApp::width = w;
61 GLApp::height = h;
62 GLApp::title = t;
63
64 if (!glfwInit()) {
65 std::cout << "GLFW init has failed - abort program!!!" << std::endl;
66 return false;
67 }
68 // In case a GLFW function fails, an error is reported to callback function
69 glfwSetErrorCallback(GLApp::error_cb);
70
71 // specify OpenGL version 4.5
72 glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, GL_MAJOR);
73 glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, GL_MINOR);
74 // specify modern OpenGL only - no compatibility with "old" OpenGL
75 glfwWindowHint(GLFW_OPENGL_PROFILE, GLFW_OPENGL_CORE_PROFILE);
76 // applications will be double-buffered ...
77 glfwWindowHint(GLFW_DOUBLEBUFFER, GLFW_TRUE);
78 // default behavior: colorbuffer is 32-bit RGBA, depthbuffer is 24-bits
79 // don't change size of window
80 glfwWindowHint(GLFW_RESIZABLE, GL_FALSE);
81
82 // size of viewport: width x height
83 GLApp::ptr_window = glfwCreateWindow(width, height, title.c_str(), NULL, NULL);
84 if (!GLApp::ptr_window) {
85 std::cerr << "GLFW unable to create OpenGL context - abort program\n";
86 glfwTerminate();
87 return false;
88 }
89
90 // make the previously created OpenGL context current ...
91 glfwMakeContextCurrent(GLApp::ptr_window);
92
93 // set callback for events associated with window size changes; keyboard;
94 // mouse buttons, cursor position, and scroller
95 glfwSetFramebufferSizeCallback(GLApp::ptr_window, GLApp::fbsize_cb);
96 glfwSetKeyCallback(GLApp::ptr_window, GLApp::key_cb);
97 glfwSetMouseButtonCallback(GLApp::ptr_window, GLApp::mousebutton_cb);
98 glfwSetCursorPosCallback(GLApp::ptr_window, GLApp::mousepos_cb);
99 glfwSetScrollCallback(GLApp::ptr_window, GLApp::mousescroll_cb);
100
101 // this is the default setting ...
102 glfwSetInputMode(GLApp::ptr_window, GLFW_CURSOR, GLFW_CURSOR_NORMAL);
103
104 // initialize OpenGL (and extension) function loading library
105 GLenum err = glewInit();
106 if (GLEW_OK != err) {
107 std::cerr << "Unable to initialize GLEW - error: "
108 << glewGetErrorString(err) << " abort program" << std::endl;
109 return false;
110 }
111 std::cout << "Using glew version: " << glewGetString(GLEW_VERSION) << std::endl;
112 std::cout << "OpenGL version: " << glGetString(GL_VERSION) << std::endl;
113
114 const GLubyte* glVersion = glGetString(GL_VERSION);
115 std::cout << "OpenGL Version String: " << glVersion << std::endl;
116
117 int major, minor;
118 if (sscanf(reinterpret_cast<const char*>(glVersion), "%d.%d", &major, &minor) != 2) {
119 std::cerr << "Failed to parse OpenGL version string." << std::endl;
120 return false;
121 }
122
123 if (major == GL_MAJOR && minor == GL_MINOR) {
124 std::cout << "OpenGL 4.5 is supported." << std::endl;
125 }
126 else {
127 std::cerr << "OpenGL 4.5 is not supported." << std::endl;
128 return false;
129 }
130
131 // we'll use the entire window as viewport ...
132 glm::ivec2 vp_width_height(0, 0);
133 glfwGetFramebufferSize(ptr_window, &vp_width_height.x, &vp_width_height.y);
134 GLApp::fbsize_cb(ptr_window, vp_width_height.x, vp_width_height.y);
135
136 return true;
137}

392void GLApp::update_time(double fps_calc_interval) {
393 // get elapsed time (in seconds) between previous and current frames
394 static double prev_time = glfwGetTime();
395 double curr_time = glfwGetTime();
396 GLApp::delta_time = curr_time - prev_time;
397 prev_time = curr_time;
398
399 // fps calculations
400 static double count = 0.0; // number of game loop iterations
401 static double start_time = glfwGetTime();
402 // get elapsed time since very beginning (in seconds) ...
403 double elapsed_time = curr_time - start_time;
404
405 ++count;
406
407 // update fps at least every 10 seconds ...
408 fps_calc_interval = (fps_calc_interval < 0.0) ? 0.0 : fps_calc_interval;
409 fps_calc_interval = (fps_calc_interval > 10.0) ? 10.0 : fps_calc_interval;
410 if (elapsed_time > fps_calc_interval) {
411 GLApp::fps = count / elapsed_time;
412 start_time = curr_time;
413 count = 0.0;
414 }
415}
