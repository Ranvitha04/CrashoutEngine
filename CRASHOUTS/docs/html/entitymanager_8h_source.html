411 template<typename T>
412 void ECS::ComponentArray<T>::RemoveComponent(Entity entityId) {
413 // quit early
414 if (entityToIndex.find(entityId) == entityToIndex.end()) return;
415
416 // reference end and current entity's value
417 size_t indexToRemove = entityToIndex.at(entityId);
418 Entity lastEntity = indexToEntity.back();
419
420 if (indexToRemove != dense.size() - 1) {
421 // Swap component data
422 dense[indexToRemove] = std::move(dense.back());
423
424 // Update the index mappings for the swapped entity
425 entityToIndex.at(lastEntity) = indexToRemove;
426 indexToEntity[indexToRemove] = lastEntity;
427 }
428
429 // Remove the entries from the sparse-set structures
430 entityToIndex.erase(entityId);
431 dense.pop_back();
432 indexToEntity.pop_back();
433 }
